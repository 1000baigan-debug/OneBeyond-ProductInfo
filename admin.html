async function matchPromoProducts() { if (!promoRows.length) return; /* IMPORTANT: The barcode in the promotional PDF is a SHORT CODE. The barcode in the weekly TXT pricing file is the ORIGINAL barcode. Product Code is the common field between both files, so Product Code is the PRIMARY matching key. We also keep the PDF barcode as a fallback only when a Product Code is not available. */ const uniqueCodes = [ ...new Set( promoRows .map(row => normaliseCode(row.code)) .filter(Boolean) ) ]; const uniqueBarcodes = [ ...new Set( promoRows .map(row => clean(row.barcode)) .filter(Boolean) ) ]; const productMapByCode = new Map(); const productMapByBarcode = new Map(); /* FIRST: Fetch products using Product Code. This is the important change. The PDF short barcode does not need to match the original pricing barcode. */ for ( let start = 0; start < uniqueCodes.length; start += PROMO_FETCH_BATCH ) { const batch = uniqueCodes.slice( start, start + PROMO_FETCH_BATCH ); const { data, error } = await supabaseClient .from("products") .select( "barcode,name,product_code,price,promo_offer,product_key" ) .in( "product_code", batch ); if (error) { throw error; } (data || []).forEach(product => { const code = normaliseCode( product.product_code ); if (!code) return; if (!productMapByCode.has(code)) { productMapByCode.set( code, [] ); } productMapByCode .get(code) .push(product); }); } /* SECOND: Fetch products by barcode as a fallback. This preserves the existing behaviour for any promotional PDF row where Product Code is missing. */ for ( let start = 0; start < uniqueBarcodes.length; start += PROMO_FETCH_BATCH ) { const batch = uniqueBarcodes.slice( start, start + PROMO_FETCH_BATCH ); const { data, error } = await supabaseClient .from("products") .select( "barcode,name,product_code,price,promo_offer,product_key" ) .in( "barcode", batch ); if (error) { throw error; } (data || []).forEach(product => { const barcode = clean(product.barcode); if (!barcode) return; if (!productMapByBarcode.has(barcode)) { productMapByBarcode.set( barcode, [] ); } productMapByBarcode .get(barcode) .push(product); }); } promoMatchedRows = []; promoRows.forEach(row => { let matched = null; /* PRIMARY MATCH: Product Code from the promo PDF against Product Code from the TXT pricing data. */ if (row.code) { const normalCode = normaliseCode( row.code ); const productsByCode = productMapByCode.get( normalCode ) || []; /* If only one database product has this Product Code, use it. */ if (productsByCode.length === 1) { matched = productsByCode[0]; } else if (productsByCode.length > 1) { /* If several products share the same Product Code, try the PDF barcode as an additional check. */ const barcodeMatches = productsByCode.filter( product => clean(product.barcode) === clean(row.barcode) ); if (barcodeMatches.length === 1) { matched = barcodeMatches[0]; } } } /* FALLBACK: If Product Code is unavailable or does not produce a unique match, try the PDF barcode against the original barcode in the database. */ if (!matched && row.barcode) { const productsByBarcode = productMapByBarcode.get( clean(row.barcode) ) || []; if (productsByBarcode.length === 1) { matched = productsByBarcode[0]; } else if ( productsByBarcode.length > 1 && row.code ) { const normalCode = normaliseCode( row.code ); matched = productsByBarcode.find( product => normaliseCode( product.product_code ) === normalCode ) || null; } } if (matched) { row.matchedProduct = matched; row.matchMethod = "PRODUCT CODE"; promoMatchedRows.push(row); } else { row.matchedProduct = null; row.matchMethod = ""; } }); document.getElementById( "promoMatched" ).textContent = promoMatchedRows.length.toLocaleString(); document.getElementById( "promoUnmatched" ).textContent = ( promoRows.length - promoMatchedRows.length ).toLocaleString(); renderPromoPreview(); }
